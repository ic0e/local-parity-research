def reverse_string(s):
    s = list(s)  # Convert string to mutable list for direct assignment
    
    left_index = 0
    right_index = len(s) - 1
    
    while left_index < right_index:
        temp = s[left_index]
        s[left_index] = s[right_index]
        s[right_index] = temp
        
        left_index += 1
        right_index -= 1
    
    return ''.join(s)

# Example usage

```

- **Swapping**: Iteratively swaps adjacent characters moving inward towards center.
- **Join Result**: After reversing character-by-character, converts back to string with `''.join(s)`.

This approach avoids reliance on non-standard libraries by leveraging basic indexing operations.

    :param s: str - Input string
    :return: Optional[str] - Index of first unique char or None

from typing import Optional
import collections

def find_first_unique(s: str) -> Optional[str]:
    """
    Returns the first non-repeating character in s. Returns None if no such character exists,
    handles empty strings/invalid types appropriately.
    
    :param s: Input string
    :return: First unique char or None
    """
    # Ensure input validity
    if not isinstance(s, str):
        raise TypeError("Expected 's' to be a string")
        
    # Early exit for empty strings
    if len(s) == 0:
        return None
    
    # Create a counter object preserving insertion order
    counter = collections.OrderedDict()
    
    # Iterate once to populate counts
    for ch in s:
        counter[ch] = counter.get(ch, 0) + 1
        
    # Find the first unique character
    for k, v in counter.items():
        if v == 1:
            return k
            
    return None

# Example usage
print(find_first_unique('google'))  # Output: g
```

import re

def parse_int(s):
    """
    Safely convert a string to an integer. Returns None if conversion fails.
    
    :param s: String representation of number
    :return: Integer value or None
    
    >>> parse_int('4')
    4
    >>> parse_int('')
    None
    >>> parse_int('a')
    None
    >>> parse_int('-5') == int('-5')
    True
    >>> parse_int('abc') is None
    True
    """
    # Check if all chars are digits including optional leading '-' sign
    matches = bool(re.match(r'^[-\d]+$', s))
    try:
        result = int(s)
    except (ValueError, TypeError):  # Catch both conversion failures
        result = None
    finally:
        return result if matches else None

# Test cases from prompt

def first_unique(s):
    # Track character frequencies
    char_counts = {}
    for ch in s:
        char_counts[ch] = char_counts.get(ch, 0) + 1
    
    # Return earliest occurrence of single-character frequency
